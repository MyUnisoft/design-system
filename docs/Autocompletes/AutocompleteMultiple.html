Custom search​

const VALUES = [
  {
    id: 1,
    label: 'Have props',
    test: 'test',
    code: 'un code',
    value: 999
  },
  {
    id: 2,
    label: 'Adam'
  },
  {
    id: 3,
    label: 'Julian'
  },
  {
    id: 4,
    label: 'Mohamed'
  }
];

const handleChange = (list) => {
  setListSelected(list);
};

const customSearch = (text, list) => {
  return list.filter((l) => l?.label?.startsWith('A'));
};

return (
  <AutocompleteMultiple
    options={VALUES}
    onChange={handleChange}
    customSearch={customSearch}
  />
);
